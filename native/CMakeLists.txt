cmake_minimum_required(VERSION 3.20)
project(xerxes_llm C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT LLAMA_CPP_DIR)
  message(FATAL_ERROR "Pass -DLLAMA_CPP_DIR=C:/path/to/llama.cpp")
endif()
add_subdirectory(${LLAMA_CPP_DIR} llama.cpp EXCLUDE_FROM_ALL)
add_executable(xerxes-llm src/xerxes_llm.c)
add_library(xerxes-llm-api SHARED src/xerxes_llm_api.c)
set_target_properties(xerxes-llm-api PROPERTIES OUTPUT_NAME xerxes-llm-api LINKER_LANGUAGE CXX)
set_target_properties(xerxes-llm PROPERTIES LINKER_LANGUAGE CXX)
target_include_directories(xerxes-llm PRIVATE ${LLAMA_CPP_DIR}/include)
target_include_directories(xerxes-llm-api PRIVATE ${LLAMA_CPP_DIR}/include)
target_link_libraries(xerxes-llm PRIVATE llama)
target_link_libraries(xerxes-llm-api PRIVATE llama)
if(MSVC)
  target_compile_options(xerxes-llm PRIVATE /W4)
else()
  target_compile_options(xerxes-llm PRIVATE -Wall -Wextra -Wpedantic)
endif()
